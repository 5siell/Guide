# 16% Luck, attackers
A <- 86
D <- 57
L <- 0.16
binomial <- c()
kills <- c()
output <- 0
kills_nr <- c()

for (i in 0:A) {
  
  binomial <- c(binomial,factorial(A)/factorial(i)/factorial(A-i)*0.6^i*0.4^(A-i))
  kills_nr <- c(kills_nr, (i-0.6*A)*L+0.6*A)
  kills <- c(kills, round((i-0.6*A)*L+0.6*A))
  
}

indices <- which(kills >= D)
new_output <- sum(binomial[indices])
binomial
kills_nr
kills
new_output*100

A <- 11
L <- 0.16
